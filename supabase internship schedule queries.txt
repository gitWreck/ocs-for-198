-- Run this whole file in Supabase SQL Editor.
-- Purpose:
-- 1. Create a database-backed FOR 198 schedule table.
-- 2. Keep schedule year in the database while allowing the portal to hide it.
-- 3. Store start/end dates for reliable "Today" highlighting.
-- 4. Store optional notes separately from the displayed date range.

create table if not exists public.internship_schedule (
    id bigint generated always as identity primary key,

    schedule_year integer not null,
    date_label text not null,
    time_range text not null default '',
    activity text not null,
    note text not null default '',

    start_date date,
    end_date date,
    sort_order integer not null,
    is_active boolean not null default true,

    created_at timestamptz not null default now(),
    updated_at timestamptz not null default now(),

    constraint internship_schedule_year_valid
        check (schedule_year between 2000 and 2100),

    constraint internship_schedule_date_range_valid
        check (
            start_date is null
            or end_date is null
            or end_date >= start_date
        ),

    constraint internship_schedule_sort_order_unique
        unique (schedule_year, sort_order)
);

create or replace function public.set_updated_at()
returns trigger
language plpgsql
as $$
begin
  new.updated_at = now();
  return new;
end;
$$;

drop trigger if exists trg_internship_schedule_updated_at
on public.internship_schedule;

create trigger trg_internship_schedule_updated_at
before update on public.internship_schedule
for each row
execute function public.set_updated_at();

alter table public.internship_schedule enable row level security;

drop policy if exists "Allow public schedule read"
on public.internship_schedule;

create policy "Allow public schedule read"
on public.internship_schedule
for select
to anon, authenticated
using (is_active = true);
